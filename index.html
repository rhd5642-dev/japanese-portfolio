// 구글 앱스 스크립트 (Code.gs) 예시 코드 function doPost(e) { try { // 프론트엔드에서 텍스트(JSON string) 형태로 보낸 데이터를 파싱 var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("기록"); // 기록을 남길 시트 이름 // 일괄 부여(액션) 요청인 경우 if (data.action === "addRecordsBulk") { var names = data.names; // ['홍길동', '김철수', '이영희'] // 이름 수만큼 시트에 한 줄씩 기록 추가 for (var i = 0; i < names.length; i++) { sheet.appendRow([ data.date, data.session, names[i], data.category, data.score || "" ]); } // 성공 반환 return ContentService.createTextOutput(JSON.stringify({ success: true, message: "저장 완료" })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
